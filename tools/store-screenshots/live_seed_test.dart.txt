// 사용법: 이 파일을 test/zz_live_seed_tmp_test.dart 로 복사해서
//   flutter test test/zz_live_seed_tmp_test.dart
// 를 돌리고, 끝나면 그 파일을 지운다 (커밋하지 말 것).
// 첫 실행 상태(키워드 미저장)에서 실제 네트워크로 확인을 돌려 저장 결과를
// OUT 경로에 JSON으로 덤프한다. seed_to_localstorage.py 가 이것을 웹 빌드용으로 바꾼다.
import 'dart:convert';
import 'dart:io';

import 'package:flutter_test/flutter_test.dart';
import 'package:news_alarm/services/news_check_service.dart';
import 'package:shared_preferences/shared_preferences.dart';

const String out = '/tmp/live_seed.json';

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();
  test('live seed', () async {
    HttpOverrides.global = null; // 테스트 바인딩의 가짜 HTTP를 끈다.
    SharedPreferences.setMockInitialValues({});
    final r = await NewsCheckService.instance.runCheck(notify: false);
    final prefs = await SharedPreferences.getInstance();
    final dump = <String, Object?>{};
    for (final k in prefs.getKeys()) {
      dump[k] = prefs.get(k);
    }
    File(out).writeAsStringSync(jsonEncode(dump));
    // ignore: avoid_print
    print('LIVE tried=${r.feedsTried} ok=${r.feedsOk} new=${r.newArticles.length}');
  }, timeout: const Timeout(Duration(minutes: 3)));
}
